async function loadContainers(queryParam = "") { try { let currentRole = (localStorage.getItem("currentUserRole") || localStorage.getItem("role") || "").trim().toLowerCase(); let userShippingLine = (localStorage.getItem("shippingLine") || localStorage.getItem("currentUser") || "").trim(); let selectedFilterLine = document.getElementById('cmbShippingLine').value; let queryParams = new URLSearchParams(); if (currentRole === "shipping_line" && userShippingLine) { queryParams.set('shipping_line', userShippingLine); document.getElementById('cmbShippingLine').value = userShippingLine; } else if (selectedFilterLine) { queryParams.set('shipping_line', selectedFilterLine); } if (queryParam) { let params = new URLSearchParams(queryParam); for (let [key, value] of params.entries()) queryParams.set(key, value); } let url = `${API_BASE}/api/containers?` + queryParams.toString(); let res = await fetch(url); let json = await res.json(); if (json.data && Array.isArray(json.data)) containerDataList = json.data; else if (Array.isArray(json)) containerDataList = json; else containerDataList = []; renderTable(containerDataList); } catch (err) { console.error("Error loading containers:", err); } } function renderTable(data) { let tbody = document.getElementById('tableBody'); tbody.innerHTML = ""; let countInYard = 0; document.getElementById('btnRelease').disabled = true; document.getElementById('btnEditBooking').disabled = true; data.forEach((item, index) => { let rawDateIn = item.date_in || item.date_in_time || item.DATE_IN || item.DATE_IN_TIME || item.datein || ""; let rawDateOut = item.date_out || item.date_out_time || item.DATE_OUT || ""; let dateInStr = rawDateIn ? String(rawDateIn).replace("T", " ").replace("Z", "").substring(0, 19) : ""; let dateOutStr = rawDateOut ? String(rawDateOut).replace("T", " ").replace("Z", "").substring(0, 19) : ""; let dayYard = 0; if (dateInStr) { let dIn = new Date(dateInStr); dIn.setHours(0, 0, 0, 0); let dEnd = new Date(); let statusVal = (item.status || "").toUpperCase(); if (statusVal.includes("RELEASE") && rawDateOut) { let dOut = new Date(dateOutStr); if (!isNaN(dOut.getTime())) { dEnd = dOut; } } dEnd.setHours(0, 0, 0, 0); if (!isNaN(dIn.getTime())) { let diffTime = dEnd - dIn; let diffDays = Math.floor(diffTime / (1000 * 60 * 60 * 24)); dayYard = diffDays >= 0 ? diffDays : 0; } } // 🟢 រាប់ចំនួនកុងតឺន័រដែលស្ថិតក្នុងស្ថានភាព IN YARD ត្រឹមត្រូវ let statusText = (item.status || "").trim().toUpperCase(); if (statusText === "IN YARD") { countInYard++; } let tr = document.createElement('tr'); tr.onclick = () => selectRow(tr, item.id, item.container_no, index); tr.innerHTML = ` ${index + 1} ${item.container_no || ''} ${dateInStr} ${dateOutStr} ${item.size || ''} ${item.type || ''} ${item.shipping_line || ''} ${item.location || '-'} ${item.status || ''} ${dayYard} ${item.vessel_voy || ''} ${item.check_repair || ''} ${item.booking_no || ''} ${item.remark || ''} `; tbody.appendChild(tr); }); // 🟢 បង្ហាញតម្លៃសរុបចូលប្រអប់ Total Con វិញ document.getElementById('txtTotalCon').innerText = countInYard; }
